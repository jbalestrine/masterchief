2️⃣ AZURE DEVOPS PIPELINE (terraform.yml)

trigger:
  branches:
    include:
      - main
pool:
  vmImage: ubuntu-latest
variables:
  TF_VERSION: "1.6.6"
steps:
- task: TerraformInstaller@1
  inputs:
    terraformVersion: $(TF_VERSION)
- script: |
    terraform init \
      -backend-config="key=$(TF_VAR_environment).tfstate"
displayName: "Terraform Init"
- script: terraform validate
  displayName: "Terraform Validate"
- script: terraform plan -out=tfplan
  displayName: "Terraform Plan"
- script: terraform apply -auto-approve tfplan
  displayName: "Terraform Apply"
📌 Why:
	• Environment-specific state
	• No secrets in YAML
	• Idempotent & auditable

3️⃣ AZURE POLICY (ENFORCEMENT)
policies/no-public-access.json

{
  "policyRule": {
    "if": {
      "anyOf": [
        { "field": "Microsoft.Sql/servers/publicNetworkAccess", "equals": "Enabled" },
        { "field": "Microsoft.Web/sites/publicNetworkAccess", "equals": "Enabled" },
        { "field": "Microsoft.Storage/storageAccounts/publicNetworkAccess", "equals": "Enabled" }
      ]
    },
    "then": {
      "effect": "deny"
    }
  }
}
Enforces no public access — even if someone messes up Terraform.

network.tf

resource "azurerm_virtual_network" "this" {
  name                = "${var.name}-vnet"
  location            = var.location
  resource_group_name = azurerm_resource_group.this.name
address_space = ["10.50.0.0/16"]
  # MODIFY if overlapping with on-prem or hub VNET
}
resource "azurerm_subnet" "app" {
  name                 = "app-subnet"
  resource_group_name  = azurerm_resource_group.this.name
  virtual_network_name = azurerm_virtual_network.this.name
  address_prefixes     = ["10.50.1.0/24"]
}
resource "azurerm_subnet" "pep" {
  name                 = "pep-subnet"
  resource_group_name  = azurerm_resource_group.this.name

appservice.tf

resource "azurerm_service_plan" "this" {
  name                = "${var.name}-plan"
  location            = var.location
  resource_group_name = azurerm_resource_group.this.name
os_type  = "Linux"
  sku_name = "P1v3"
  # Increase for prod workloads
}
resource "azurerm_linux_web_app" "this" {
  name                = "${var.name}-app"
  location            = var.location
  resource_group_name = azurerm_resource_group.this.name
  service_plan_id     = azurerm_service_plan.this.id
public_network_access_enabled = false
  # Forces traffic through Private Endpoint
identity {
    type         = "UserAssigned"
    identity_ids = [azurerm_user_assigned_identity.app.id]
  }
app_settings = {
    "AZURE_SQL_AUTH" = "ManagedIdentity"
    # App code must use AAD auth
  }
}
