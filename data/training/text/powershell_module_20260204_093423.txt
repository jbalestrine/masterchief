function Invoke-DMSDeploy {
    <#
    .SYNOPSIS
        Deploy Azure environment with standardized naming
    .EXAMPLE
        Invoke-DMSDeploy -Environment prod -Location eastus -Instance 001
    #>
    [CmdletBinding()]
    param(
        [Parameter(Mandatory=$true)]
        [ValidateSet('dev','test','staging','prod')]
        [string]$Environment,
        
        [Parameter(Mandatory=$false)]
        [string]$Location = $Script:DMSConfig.DefaultRegion,
        
        [Parameter(Mandatory=$true)]
        [ValidatePattern('^\d{3}$')]
        [string]$Instance
    )
    
    Write-Host "Deploying environment: $Environment-$Location-$Instance" -ForegroundColor Cyan
    
    $scriptPath = Join-Path $PSScriptRoot "..\..\Scripts\Deployment\Deploy-AzureEnvironment.ps1"
    if(Test-Path $scriptPath) {
        & $scriptPath -Environment $Environment -Location $Location -Instance $Instance
    } else {
        Write-Error "Deployment script not found: $scriptPath"
    }
}

function Get-DMSInventory {
    <#
    .SYNOPSIS
        Generate comprehensive Azure resource inventory
    .EXAMPLE
        Get-DMSInventory -ExportPath "C:\inventory.json"
    #>
    [CmdletBinding()]
    param(
        [Parameter(Mandatory=$false)]
        [string]$SubscriptionId,
        
        [Parameter(Mandatory=$false)]
        [string]$ExportPath
    )
    
    Write-Host "Generating Azure inventory..." -ForegroundColor Cyan
    
    $scriptPath = Join-Path $PSScriptRoot "..\..\Scripts\Management\Get-AzureInventory.ps1"
    if(Test-Path $scriptPath) {
        $params = @{}
        if($SubscriptionId) { $params.SubscriptionId = $SubscriptionId }
        if($ExportPath) { $params.OutputPath = $ExportPath }
        
        & $scriptPath @params
    } else {
        Write-Error "Inventory script not found: $scriptPath"
    }
}

function Remove-DMSEnvironment {

function Start-DMSResources {
    <#
    .SYNOPSIS
        Start stopped Azure resources
    .EXAMPLE
        Start-DMSResources -ResourceGroupName "dev-rg-eus-001"
    #>
    [CmdletBinding()]
    param(
        [Parameter(Mandatory=$true)]
        [string]$ResourceGroupName
    )
    
    Write-Host "Starting resources in: $ResourceGroupName" -ForegroundColor Cyan
    
    try {
        $vms = Get-AzVM -ResourceGroupName $ResourceGroupName -Status | Where-Object { $_.PowerState -eq 'VM deallocated' }
        foreach($vm in $vms) {
            Write-Host "  Starting VM: $($vm.Name)" -ForegroundColor Yellow
            Start-AzVM -ResourceGroupName $ResourceGroupName -Name $vm.Name -NoWait
        }
        Write-Host "✓ Start commands issued for $($vms.Count) VMs" -ForegroundColor Green
    } catch {
        Write-Error "Failed to start resources: $_"
    }
}

function Stop-DMSResources {
    <#
    .SYNOPSIS
        Stop running Azure resources to save costs
    .EXAMPLE
        Stop-DMSResources -ResourceGroupName "dev-rg-eus-001"
    #>
    [CmdletBinding(SupportsShouldProcess, ConfirmImpact='High')]
    param(
        [Parameter(Mandatory=$true)]
        [string]$ResourceGroupName,
        
        [Parameter(Mandatory=$false)]
        [switch]$Force
    )
    
    if($PSCmdlet.ShouldProcess($ResourceGroupName, "Stop resources")) {
        Write-Host "Stopping resources in: $ResourceGroupName" -ForegroundColor Yellow
        
        try {
            $vms = Get-AzVM -ResourceGroupName $ResourceGroupName -Status | Where-Object { $_.PowerState -eq 'VM running' }
            foreach($vm in $vms) {
                Write-Host "  Stopping VM: $($vm.Name)" -ForegroundColor Yellow
                Stop-AzVM -ResourceGroupName $ResourceGroupName -Name $vm.Name -Force:$Force -NoWait
            }
            Write-Host "✓ Stop commands issued for $($vms.Count) VMs" -ForegroundColor Green
        } catch {
            Write-Error "Failed to stop resources: $_"
        }
    }
}

function Test-DMSConnection {
    <#
    .SYNOPSIS
        Test Azure connection and DMS configuration
    .EXAMPLE
        Test-DMSConnection
    #>
    [CmdletBinding()]
    param()
    
    Write-Host "Testing DevOps Master Suite connectivity..." -ForegroundColor Cyan
    Write-Host ""
    
    # Test Azure connection
    Write-Host "[1/3] Azure Connection..." -NoNewline
    try {
        $context = Get-AzContext
        if($context) {
            Write-Host " ✓" -ForegroundColor Green
            Write-Host "  Subscription: $($context.Subscription.Name)" -ForegroundColor Gray
            Write-Host "  Account: $($context.Account.Id)" -ForegroundColor Gray
        } else {
            Write-Host " ✗" -ForegroundColor Red
            Write-Host "  Run 'Connect-AzAccount' to authenticate" -ForegroundColor Yellow
        }
    } catch {
        Write-Host " ✗" -ForegroundColor Red
    }
    
    # Test API endpoint
    Write-Host ""
    Write-Host "[2/3] API Server..." -NoNewline
    try {
        $response = Invoke-RestMethod -Uri "$($Script:DMSConfig.APIEndpoint)/api/health" -ErrorAction Stop
        Write-Host " ✓" -ForegroundColor Green
        Write-Host "  Endpoint: $($Script:DMSConfig.APIEndpoint)" -ForegroundColor Gray
        Write-Host "  Status: $($response.status)" -ForegroundColor Gray
    } catch {
        Write-Host " ✗" -ForegroundColor Red
        Write-Host "  API server not running. Run 'start-backend.ps1'" -ForegroundColor Yellow
    }
    
    # Test scripts directory
    Write-Host ""
    Write-Host "[3/3] Scripts Directory..." -NoNewline
    $scriptsPath = Join-Path $PSScriptRoot "..\..\Scripts"
    if(Test-Path $scriptsPath) {
        $scriptCount = (Get-ChildItem -Path $scriptsPath -Recurse -Filter "*.ps1").Count
        Write-Host " ✓" -ForegroundColor Green
        Write-Host "  Location: $scriptsPath" -ForegroundColor Gray
        Write-Host "  Scripts found: $scriptCount" -ForegroundColor Gray
    } else {
        Write-Host " ✗" -ForegroundColor Red
    }
    
    Write-Host ""
}
