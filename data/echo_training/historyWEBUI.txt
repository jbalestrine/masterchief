# command_mode.py

class CommandMode:
    def __init__(self, session, chat_ui, ingestion_model):
        self.session = session
        self.chat_ui = chat_ui
        self.model = ingestion_model
        self.active = False
        self.motd = "Welcome to Command Mode! Type !help for commands."
        self.commands = {}  # command_name: function
        self.aliases = {}   # optional
        self.permissions = {}  # user_role: allowed commands
        self.logs = []

    def toggle(self, enable: bool):
        self.active = enable
        if enable:
            self.chat_ui.display_motd(self.motd)

    def register_command(self, name: str, func, required_role: str = "all"):
        self.commands[name] = func
        self.permissions[name] = required_role

    def execute(self, user, message):
        if not self.active or not message.startswith("!"):
            return None
        cmd_name = message[1:].split()[0]
        args = message.split()[1:]
        # Check alias
        if cmd_name in self.aliases:
            cmd_name = self.aliases[cmd_name]
        # Check permission
        role = self.session.get_user_role(user)
        if role not in ["SOP", "OPS", "all"] and self.permissions.get(cmd_name, "all") != "all":
            return f"Permission denied for {cmd_name}"
        # Execute
        func = self.commands.get(cmd_name)
        if not func:
            return f"Unknown command {cmd_name}"
        # Run command via ingestion/training model
        result = func(args, self.session, self.model)
        # Log
        self.logs.append({"user": user, "cmd": cmd_name, "args": args})
        return result

3️⃣ Backend Example (Python Flask)
from flask import Flask, request, jsonify
import json
import subprocess
import uuid
import os

app = Flask(__name__)

TEMPLATES_DIR = "./ArmTemplates"

@app.route("/deploy-arm", methods=["POST"])
def deploy_arm():
    data = request.json

    # Generate unique parameters file
    deploy_id = str(uuid.uuid4())
    param_file_path = os.path.join(TEMPLATES_DIR, f"params_{deploy_id}.json")
    
    with open(param_file_path, "w") as f:
        json.dump(data, f, indent=2)

    # Call Azure CLI deployment (non-blocking)
    rg = data.get("resourceGroupName", "EnterpriseAppRG")
    template_file = "./arm-template.json"

    cmd = [
        "az", "deployment", "group", "create",
        "--resource-group", rg,
        "--template-file", template_file,
        "--parameters", f"@{param_file_path}",
        "--verbose"
    ]

    # Run deployment asynchronously
    process = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    
    return jsonify({
        "status": "deployment_started",
        "deploy_id": deploy_id
    })

if __name__ == "__main__":
    app.run(port=8080)
