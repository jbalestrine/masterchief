const express = require('express');
const cors = require('cors');
const fs = require('fs');
const path = require('path');
const { exec } = require('child_process');

const app = express();
app.use(express.json());
app.use(cors());

// Load feature flag
const config = JSON.parse(fs.readFileSync(path.join(__dirname, '../config.json'), 'utf8'));
const ultraMode = config.ultra_mode_enabled;

// List of core and ultra features
const coreFeatures = ["powershell", "hyperv", "azure", "terraform", "vde", "cicd", "vault"];
const ultraFeatures = [
  "docker","aks","vmware","drift","rollback","sched","prometheus","grafana","cms","game",
  "plugins","multicloud","api_gateway","dns","irc","discord","chatgpt","learnengine",
  "sketch","templates","integrations","logs"
];

// Combined features list based on flag
const allFeatures = ultraMode ? coreFeatures.concat(ultraFeatures) : coreFeatures;

// Endpoint for each feature
allFeatures.forEach(feature => {
  app.post('/run-' + feature, (req, res) => {
    const useDefaults = req.body.useDefaults;
    const param1 = req.body.param1 || feature + "_default1";
    const param2 = req.body.param2 || feature + "_default2";

    const scriptPath = path.join(__dirname, '../scripts', `${feature}_script.ps1`);
    if (!fs.existsSync(scriptPath)) {
      res.send(`Error: Script not found for ${feature}`);
      return;
    }

    let command;
    if (useDefaults) {
      command = `powershell -ExecutionPolicy Bypass -File "${scriptPath}"`;
    } else {
      command = `powershell -ExecutionPolicy Bypass -File "${scriptPath}" -Param1 "${param1}" -Param2 "${param2}"`;
    }

    exec(command, { shell: 'powershell.exe' }, (error, stdout, stderr) => {
      if (error) {
        res.send(`Error:\n${stderr || error.message}`);
      } else {
        res.send(`Output:\n${stdout}`);
      }
    });
  });
});

// Extra endpoints for Terraform
app.post('/generate-tf', (req, res) => {
  const { rg, location, vmname, vmsize, vnet } = req.body;
  const tfContent = `
provider "azurerm" { features {} }

resource "azurerm_resource_group" "example" {
  name     = "${rg}"
  location = "${location}"
}

resource "azurerm_virtual_network" "example" {
  name                = "${vnet}"
  address_space       = ["10.0.0.0/16"]
  location            = azurerm_resource_group.example.location
  resource_group_name = azurerm_resource_group.example.name
}

resource "azurerm_linux_virtual_machine" "example" {
  name                = "${vmname}"
  resource_group_name = azurerm_resource_group.example.name
  location            = azurerm_resource_group.example.location
  size                = "${vmsize}"
  admin_username      = "adminuser"
  disable_password_authentication = true
  network_interface_ids = []
}
  `;
  const filePath = path.join(__dirname, '../scripts/generated_config.tf');
  fs.writeFileSync(filePath, tfContent);
  res.send(tfContent);
});

app.post('/deploy-tf', (req, res) => {
  const scriptPath = path.join(__dirname, '../scripts');
  exec('terraform init && terraform apply -auto-approve', { cwd: scriptPath }, (error, stdout, stderr) => {
    if (error) {
      res.send(`Error:\n${stderr || error.message}`);
    } else {
      res.send(`Output:\n${stdout}`);
    }
  });
});

app.listen(3000, () => {
  console.log('Node backend running at http://localhost:3000');
});
