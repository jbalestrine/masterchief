function Show-EnterpriseMenu {
    Clear-Host
    Write-Host "=== MasterChief Enterprise Menu ===" -ForegroundColor Cyan
    Write-Host "1. Install Dev Tools"
    Write-Host "2. Install Docker + WSL2"
    Write-Host "3. Azure Login"
    Write-Host "4. System Status"
    Write-Host "5. Exit"
}

function Install-DevTools {
    winget install --id Microsoft.VisualStudioCode -e
    winget install --id Microsoft.SQLServerManagementStudio -e
    winget install --id Git.Git -e
    winget install --id Notepad++.Notepad++ -e
}

function Azure-Login {
    az login
}

Set-Alias emenu Show-EnterpriseMenu
